# Autosave while linked (lab saves, Route 101, linked, Beacon ON). Each
# save autosaves on its own: Rosa (player 0) walks into Oldale Town and
# autosaves while Kai stands on Route 101; then Kai follows and autosaves.
# A full save holds the game for about four seconds of flash time, so this
# checks the link survives it on both sides. FIXTURES: the 30 s interval.
# Checks: each player's "Autosaving..." and one autosave each; afterwards
# both are still linked and see each other; each file's newest slot is an
# autosave and the other its manual save.
include wild_route101_start.inc
b no_wild 1
b write gCoopAutosaveTestInterval:4 1800
b wait 1900
0 goto 11 2 2000
0 note_text Autosaving
0 cross UP 0 10 4000
0 until_mem gCoopAutosaveCount 1 1500
0 mash_field 600 B
0 expect_noted
1 expect gCoopAutosaveCount 0
b wait 120
b expect coop.sLinkState 2
1 goto 11 2 2000
1 note_text Autosaving
1 cross UP 0 10 4000
1 until_mem gCoopAutosaveCount 1 1500
1 mash_field 600 B
1 expect_noted
b wait 120
b expect coop.sLinkState 2
b expect_partner oldale
0 expect gCoopAutosaveCount 1
b hex gSaveCounter 4
