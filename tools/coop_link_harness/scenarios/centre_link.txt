# LINK UP after a Pokemon Center visit made while unlinked (story_seg6
# saves, Rustboro). Gameplay test, no fixtures beyond no_wild.
# Found in the main journey (segment 06): a save continued inside a Center
# never linked ("Cable players: 0"). The Center's wireless Union Room scan,
# run while unlinked, left the RFU serial/timer handlers in place and the
# comm type wireless, also after leaving; the cable handshake then never
# saw a serial interrupt.
#  1. Both unlinked: into the Centre, heal, out; LINK UP outdoors.
#  2. Linked: into the Centre, heal; UNLINK there; LINK UP inside.
#  3. Out again, still linked.
include continue.inc
b no_wild 1
b expect coop.sLinkState 0
0 walk 16 38 4000
0 until_map 11 5 3000
1 wait 120
1 walk 16 38 4000
1 until_map 11 5 3000
include journey_heal_center.inc
b hex gWirelessCommType 1
b walk 7 8 2000
b cross DOWN 0 3 2000
b wait 60
include relink.inc
b expect coop.sLinkState 2
b expect gWirelessCommType 0
b shot linked-after-centre
# 2. Back in linked; heal; unlink and relink inside.
1 walk 16 38 4000
1 until_map 11 5 3000
0 wait 120
0 walk 16 38 4000
0 until_map 11 5 3000
include journey_heal_center.inc
b expect coop.sLinkState 2
0 walk 6 5 2000
1 walk 8 5 2000
b sync
include unlink.inc
b expect coop.sLinkState 0
b wait 120
include relink.inc
b expect coop.sLinkState 2
b shot linked-inside-centre
b walk 7 8 2000
b cross DOWN 0 3 2000
b wait 120
b expect coop.sLinkState 2
b dump end
